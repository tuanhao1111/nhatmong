// ════════════════════════════════════════════════════════════════════════════
//  FIRESTORE SECURITY RULES — Nhất Mộng (project: nhatmongdata)
// ════════════════════════════════════════════════════════════════════════════
//
//  CÁCH DÙNG:
//    1. Firebase Console → Firestore Database → Rules
//    2. Copy toàn bộ block `rules_version=...` bên dưới (KHÔNG copy header này)
//    3. Publish
//
//  PREREQUISITE:
//    - Authentication → Sign-in method → Email/Password: ENABLED
//    - Authentication → Sign-in method → Anonymous: ENABLED
//      (Bắt buộc cho guest đọc data — nếu tắt, nút "Vào với tư cách Khách"
//       sẽ báo auth/admin-restricted-operation)
//
//  BOOTSTRAP ADMIN ĐẦU TIÊN:
//    1. Đăng ký account `admin` qua login.html (nó sẽ tạo role='member')
//    2. Vào Firebase Console → Firestore → /users/{uid của admin}
//    3. Sửa field role: "admin"
//    4. Reload app → admin có full quyền
//
// ────────────────────────────────────────────────────────────────────────────

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ── Helpers ──────────────────────────────────────────────────────────
    function isSignedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return isSignedIn() &&
             exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    function isMember() {
      return isSignedIn() &&
             exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role in ['admin', 'member'];
    }

    function isSelf(uid) {
      return isSignedIn() && request.auth.uid == uid;
    }

    // Validate text input — chống injection từ client compromised
    function isCleanText(s, maxLen) {
      return s is string && s.size() <= maxLen &&
             !s.matches('.*[<>"\'`\\\\\\u0000-\\u001f\\u007f].*');
    }

    // ── /users/{uid} — Profile + role ────────────────────────────────────
    // Read: tất cả người đã đăng nhập (kể cả guest anonymous) — để load
    //       danh sách user trong trang Quản Lý Tài Khoản, role lookup, ...
    // Create: chỉ user tạo doc của chính mình, role bắt buộc = 'member'
    // Update: admin sửa role; user tự sửa name/profile (KHÔNG được tự
    //         promote role)
    // Delete: chỉ admin
    match /users/{uid} {
      allow read:   if isSignedIn();

      allow create: if isSelf(uid)
                    && request.resource.data.role == 'member'
                    && request.resource.data.id == uid
                    && isCleanText(request.resource.data.name, 60)
                    && isCleanText(request.resource.data.username, 40);

      allow update: if isAdmin()
                    || (isSelf(uid)
                        && request.resource.data.role == resource.data.role  // không tự promote
                        && request.resource.data.id == resource.data.id      // không đổi UID
                        && isCleanText(request.resource.data.name, 60));

      allow delete: if isAdmin();
    }

    // ── /guilds/guild_main — settings + sessions (single doc) ────────────
    match /guilds/guild_main {
      allow read:  if isSignedIn();
      allow write: if isAdmin();

      // ── /guilds/guild_main/members/{uid} — member info ─────────────────
      // Read: signed-in (guest cũng xem được danh sách bang)
      // Write: admin · hoặc member tự sửa info bản thân (cùng UID)
      match /members/{memberId} {
        allow read:   if isSignedIn();
        allow create: if isAdmin();
        allow update: if isAdmin()
                      || (isSelf(memberId)
                          && request.resource.data.role == resource.data.role
                          && request.resource.data.id == memberId);
        allow delete: if isAdmin();
      }

      // ── /guilds/guild_main/gacha/state — Gacha cycle/rosters/rules ────
      // Read: signed-in (guest đọc được)
      // Write: admin only — push toàn bộ state qua merge:true (xem
      //        gacha-component.html GachaSync.push)
      match /gacha/{docId} {
        allow read:  if isSignedIn();
        allow write: if isAdmin();
      }

      // ── /guilds/guild_main/gacha/audit/{ts} — (FUTURE) audit log ───────
      // Append-only log của các lần sửa thể lệ. Admin write, all read.
      match /gacha/audit/{entryId} {
        allow read:   if isSignedIn();
        allow create: if isAdmin();
        allow update: if false;  // immutable
        allow delete: if isAdmin();
      }
    }

    // ── /minigames — game on/off + rooms ─────────────────────────────────
    match /minigames/status {
      allow read:  if isSignedIn();
      allow write: if isAdmin();
    }

    // Multiplayer rooms — Phase 4 (chưa implement)
    match /minigames/rooms/{roomId} {
      allow read:   if isSignedIn();
      // Người tạo phòng = host. Host và admin được sửa.
      allow create: if isSignedIn();
      allow update: if isAdmin()
                    || (isSignedIn() && resource.data.hostUid == request.auth.uid);
      allow delete: if isAdmin()
                    || (isSignedIn() && resource.data.hostUid == request.auth.uid);
    }

    // ── Catch-all: deny ──────────────────────────────────────────────────
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
